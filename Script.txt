Script Arduino IDE
#include <WiFi.h>
#include <ThingSpeak.h>
#include <DHT.h>

// Konfigurasi pin DHT22
#define DHTPIN 4
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);

// Konfigurasi WiFi
const char* ssid = "DnZO_wrt";
const char* password = "internet1";

// Konfigurasi ThingSpeak
unsigned long channelID = 3569501;
const char* writeAPIKey = "XUWGOCZFRPKKTG9N";

// Interval pengambilan data: 60 detik
const unsigned long interval = 60000;
unsigned long previousMillis = 0;

WiFiClient client;

void setup() {
  Serial.begin(115200);

  // Memulai sensor DHT22
  dht.begin();

  // Menghubungkan WiFi
  Serial.println("Menghubungkan ke WiFi...");
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi berhasil terhubung!");
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());

  // Memulai ThingSpeak
  ThingSpeak.begin(client);

  // Agar pembacaan pertama dilakukan segera
  previousMillis = millis() - interval;
}

void loop() {
  unsigned long currentMillis = millis();

  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis;

    // Membaca sensor
    float temperature = dht.readTemperature();
    float humidity = dht.readHumidity();

    // Memeriksa hasil pembacaan
    if (isnan(temperature) || isnan(humidity)) {
      Serial.println("Gagal membaca sensor DHT22!");
      return;
    }

    // Menampilkan data di Serial Monitor
    Serial.println("========================");
    Serial.print("Suhu       : ");
    Serial.print(temperature);
    Serial.println(" °C");

    Serial.print("Kelembapan : ");
    Serial.print(humidity);
    Serial.println(" %RH");

    // Mengirim data ke ThingSpeak
    ThingSpeak.setField(1, temperature);
    ThingSpeak.setField(2, humidity);

    int response = ThingSpeak.writeFields(
      channelID,
      writeAPIKey
    );

    if (response == 200) {
      Serial.println("Data berhasil dikirim ke ThingSpeak!");
    } else {
      Serial.print("Gagal mengirim data. Kode: ");
      Serial.println(response);
    }

    Serial.println("========================");
  }
}

#include <WiFi.h>
#include <HTTPClient.h>
#include <time.h>

const char* ssid = "Wokwi-GUEST";
const char* password = "";

// Ganti dengan Write API Key channel ThingSpeak Anda
const char* writeApiKey = "XUWGOCZFRPKKTG9N";

// Konfigurasi NTP untuk Jam Real-Time (WIB = UTC+7)
const char* ntpServer = "pool.ntp.org";
const long gmtOffset_sec = 7 * 3600; // Sesuaikan jika WITA (8) atau WIT (9)
const int daylightOffset_sec = 0;

// Fungsi helper untuk mengambil string jam real-time (HH:MM:SS)
String getRealTime() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return "00:00:00";
  }
  char buffer[10];
  strftime(buffer, sizeof(buffer), "%H:%M:%S", &timeinfo);
  return String(buffer);
}

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi terhubung!");

  // Sinkronisasi waktu NTP
  configTime(gmtOffset_sec, daylightOffset_sec, ntpServer);
  Serial.println("Menyelaraskan waktu NTP...");
  delay(2000);
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    
    static int seq = 0;
    seq++;
    
    String serverUrl = "http://api.thingspeak.com/update";
    
    // --- SKENARIO 1: HTTP POST - PAYLOAD KECIL (~20 byte) ---
    String httpRequestData = "api_key=" + String(writeApiKey) + "&field1=" + String(seq);
    // --------------------------------------------------------

    String sendTimeStr = getRealTime(); // Jam real-time saat dikirim
    unsigned long startTime = millis(); // Mulai hitung RTT
    
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/x-www-form-urlencoded");
    
    int httpResponseCode = http.POST(httpRequestData);
    
    unsigned long endTime = millis();   // Selesai hitung RTT
    unsigned long rtt = endTime - startTime;
    String receiveTimeStr = getRealTime(); // Jam real-time saat diterima
    int rssi = WiFi.RSSI();

    if (httpResponseCode > 0) {
      Serial.print("[S1] SEQ: "); Serial.print(seq);
      Serial.print(" | Kirim: "); Serial.print(sendTimeStr);
      Serial.print(" | Terima: "); Serial.print(receiveTimeStr);
      Serial.print(" | HTTP RTT: "); Serial.print(rtt);
      Serial.print(" ms | RSSI: "); Serial.print(rssi);
      Serial.print(" dBm | Code: "); Serial.println(httpResponseCode);
    } else {
      Serial.print("Error HTTP POST: "); Serial.println(httpResponseCode);
    }

    http.end();
  }
  
  delay(2000); // Jeda wajib 2 detik per pesan[cite: 5]
}

const int daylightOffset_sec = 0;

WiFiClient wifiClient;
MqttClient mqttClient(wifiClient);

int currentSeq = 0;

// Fungsi helper untuk mengambil string jam real-time (HH:MM:SS)
String getRealTime() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return "00:00:00";

void loop() {
  // Pastikan terhubung ke broker MQTT
  if (!mqttClient.connected()) {
    Serial.print("Menghubungkan ke MQTT Broker...");
    if (!mqttClient.connect(mqtt_server, mqtt_port)) {
      Serial.print("Gagal, error = ");
      Serial.println(mqttClient.connectError());
      delay(5000);
      return;
    }
    Serial.println("Terhubung!");
  }

  mqttClient.poll();

  static unsigned long lastMsgTime = 0;
  unsigned long now = millis();
  
  // Kirim pesan setiap 2 detik sesuai ketentuan eksperimen Bagian B
  if (now - lastMsgTime > 2000) {
    lastMsgTime = now;
    currentSeq++;

    // --- SKENARIO 3: MQTT QoS 0 - PAYLOAD KECIL (~20 byte) ---[cite: 5]

    // Kirim pesan dengan payload kecil dan QoS 0
    mqttClient.beginMessage("iot/uts/publish", false, 0); // (topik, retained, qos=0)
    mqttClient.print(payload);
    int success = mqttClient.endMessage(); // Eksekusi pengiriman

    unsigned long endTime = millis();
    unsigned long rtt = endTime - startTime; // RTT dalam milidetik
    String receiveTimeStr = getRealTime(); // Jam real-time saat diterima

      Serial.print("[S3] Gagal Kirim Seq: "); Serial.print(currentSeq);
